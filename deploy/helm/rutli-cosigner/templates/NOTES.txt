     kubectl -n {{ $.Release.Namespace }} create secret tls {{ $node.tlsSecret }} --cert=certs/cosigner{{ int $node.id }}.crt --key=certs/cosigner{{ int $node.id }}.key
{{- end }}

  2) Seal bundles (on a workstation; `seal new-node` for a new cluster, `seal split-legacy` to migrate):
{{- range $node := $nodes }}
     kubectl -n {{ $.Release.Namespace }} create secret generic {{ $node.sealSecret }} --from-file={{ $.Values.seal.secretKey }}=node-{{ int $node.id }}.json
{{- end }}
     then delete the local bundle files. Never keep two nodes' bundles in one place.
